function MessageBubble({ msg }) {
  const isUser = msg.role === "user";

  const renderContent = (text) => {
    const parts = text.split(/(\*\*[^*]+\*\*)/g);
    return parts.map((part, i) =>
      part.startsWith("**") && part.endsWith("**") ? (
        <strong key={i} style={{ color: "#38bdf8", fontWeight: 600 }}>
          {part.slice(2, -2)}
        </strong>
      ) : (
        <span key={i}>{part}</span>
      )
    );
  };

  return (
    <div
      style={{
        display: "flex",
        gap: 12,
        flexDirection: isUser ? "row-reverse" : "row",
        marginBottom: 20,
        animation: "fadeSlideIn 0.3s ease-out",
      }}
    >
      <div
        style={{
          width: 32,
          height: 32,
          borderRadius: "50%",
          flexShrink: 0,
          background: isUser
            ? "linear-gradient(135deg, #818cf8, #6366f1)"
            : "linear-gradient(135deg, #0ea5e9, #38bdf8)",
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          fontSize: 14,
          border: "1px solid rgba(255,255,255,0.1)",
        }}
      >
        {isUser ? "U" : "F"}
      </div>
      <div style={{ maxWidth: "75%", display: "flex", flexDirection: "column", gap: 4, alignItems: isUser ? "flex-end" : "flex-start" }}>
        <div
          style={{
            padding: "10px 16px",
            borderRadius: isUser ? "18px 4px 18px 18px" : "4px 18px 18px 18px",
            background: isUser
              ? "linear-gradient(135deg, rgba(99,102,241,0.25), rgba(129,140,248,0.15))"
              : "rgba(15,23,42,0.6)",
            border: isUser
              ? "1px solid rgba(129,140,248,0.3)"
              : "1px solid rgba(56,189,248,0.15)",
            backdropFilter: "blur(8px)",
            color: "#e2e8f0",
            fontSize: 14,
            lineHeight: 1.65,
            letterSpacing: "0.01em",
            fontFamily: "'Sora', sans-serif",
            whiteSpace: "pre-wrap",
          }}
        >
          {renderContent(msg.content)}
        </div>
        <span style={{ fontSize: 11, color: "#475569", letterSpacing: "0.05em" }}>{msg.time}</span>
      </div>
    </div>
  );
}

function StatBar({ label, value, unit }) {
  const color = value > 80 ? "#f87171" : value > 60 ? "#fb923c" : "#38bdf8";
  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 4 }}>
      <div style={{ display: "flex", justifyContent: "space-between", fontSize: 11, color: "#64748b", letterSpacing: "0.08em" }}>
        <span>{label}</span>
        <span style={{ color }}>{value}{unit}</span>
      </div>
      <div style={{ height: 3, background: "rgba(255,255,255,0.05)", borderRadius: 2 }}>
        <div
          style={{
            height: "100%",
            width: `${value}%`,
            borderRadius: 2,
            background: `linear-gradient(90deg, ${color}88, ${color})`,
            transition: "width 1s ease",
          }}
        />
      </div>
    </div>
  );
}

              <span style={{ fontSize: 11, color: currentColor, letterSpacing: "0.1em", fontWeight: 500 }}>
                {mode.toUpperCase()} MODE ACTIVE
              </span>
              <span style={{ fontSize: 11, color: "#334155", marginLeft: "auto" }}>
                {messages.filter(m => m.role === "user").length} interactions
              </span>
            </div>

            {/* Messages */}
            <div
              style={{
                flex: 1,
                overflowY: "auto",
                padding: "24px",
                display: "flex",
                flexDirection: "column",
              }}
            >
              {messages.map((msg) => (
                <MessageBubble key={msg.id} msg={msg} />
              ))}
              {loading && (
                <div style={{ display: "flex", gap: 12, marginBottom: 20, animation: "fadeSlideIn 0.3s ease-out" }}>
                  <div
                    style={{
                      width: 32,
                      height: 32,
                      borderRadius: "50%",
                      background: "linear-gradient(135deg, #0ea5e9, #38bdf8)",
                      display: "flex",
                      alignItems: "center",
                      justifyContent: "center",
                      fontSize: 14,
                      flexShrink: 0,
                    }}
                  >
                    F
                  </div>
                  <div
                    style={{
                      background: "rgba(15,23,42,0.6)",
                      border: "1px solid rgba(56,189,248,0.15)",
                      borderRadius: "4px 18px 18px 18px",
                      backdropFilter: "blur(8px)",
                    }}
                  >
                    <TypingIndicator />
                  </div>
                </div>
              )}
              <div ref={bottomRef} />
            </div>

            {/* INPUT AREA */}
            <div
              style={{
                padding: "16px 24px",
                borderTop: "1px solid rgba(56,189,248,0.08)",
                background: "rgba(2,8,23,0.7)",
                backdropFilter: "blur(20px)",
                flexShrink: 0,
              }}
            >
              <div
                style={{
                  display: "flex",
                  gap: 10,
                  alignItems: "flex-end",
                  background: "rgba(15,23,42,0.8)",
                  border: `1px solid ${input ? "rgba(56,189,248,0.3)" : "rgba(56,189,248,0.1)"}`,
                  borderRadius: 14,
                  padding: "10px 14px",
                  backdropFilter: "blur(10px)",
                  transition: "border-color 0.2s",
                  boxShadow: input ? "0 0 20px rgba(56,189,248,0.08)" : "none",
                }}
              >
                <textarea
                  ref={inputRef}
                  value={input}
                  onChange={(e) => setInput(e.target.value)}
                  onKeyDown={handleKey}
                  placeholder={`Message Fuad in ${mode} mode...`}
                  rows={1}
                  style={{
                    flex: 1,
                    color: "#e2e8f0",
                    fontSize: 14,
                    fontFamily: "'Sora', sans-serif",
                    lineHeight: 1.6,
                    maxHeight: 120,
                    overflowY: "auto",
                    letterSpacing: "0.01em",
                  }}
                  onInput={(e) => {
                    e.target.style.height = "auto";
                    e.target.style.height = Math.min(e.target.scrollHeight, 120) + "px";
                  }}
                />
                <div style={{ display: "flex", gap: 6, alignItems: "center" }}>
                  <button
                    onClick={toggleListening}
                    className="action-btn"
                    style={{
                      width: 34,
                      height: 34,
                      borderRadius: 9,
                      background: listening ? "rgba(244,114,182,0.15)" : "rgba(255,255,255,0.04)",
                      border: `1px solid ${listening ? "rgba(244,114,182,0.3)" : "rgba(255,255,255,0.07)"}`,
                      cursor: "pointer",
                      fontSize: 15,
                      display: "flex",
                      alignItems: "center",
                      justifyContent: "center",
                      transition: "all 0.2s",
                      flexShrink: 0,
                    }}
                  >
                    🎙
                  </button>
                  <button
                    className="send-btn"
                    onClick={() => sendMessage(input)}
                    disabled={!input.trim() || loading}
                    style={{
                      width: 34,
                      height: 34,
                      borderRadius: 9,
                      background: input.trim() && !loading
                        ? "linear-gradient(135deg, #0ea5e9, #6366f1)"
                        : "rgba(255,255,255,0.04)",
                      border: "none",
                      cursor: input.trim() && !loading ? "pointer" : "not-allowed",
                      color: input.trim() && !loading ? "#fff" : "#334155",
                      fontSize: 15,
                      display: "flex",
                      alignItems: "center",
                      justifyContent: "center",
                      transition: "all 0.2s",
                      flexShrink: 0,
                    }}
                  >
                    ↑
                  </button>
                </div>
              </div>
              <div style={{ display: "flex", justifyContent: "center", marginTop: 8 }}>
                <span style={{ fontSize: 10, color: "#1e293b", letterSpacing: "0.08em" }}>
                  FUAD v2.0 · {mode} Mode · Enter to send · Shift+Enter for new line
                </span>
              </div>
            </div>
          </div>
        </div>
      </div>
    </>
  );
}
